// حداکثر سرعت مجاز اسپیندل
float getMaxSpindleRPM()
{
  if (GEAR_RATIO <= 0.0)
    return 0.0;

  return SERVO_MAX_RPM / GEAR_RATIO;

}

// دور اسپیندل مورد نیاز برای عمق
float calculateSpindleRevolutions(float depth, float pitch)
{
  if (pitch <= 0.0)
    return 0.0;

  return depth / pitch;
}

// دور موتور سروو مورد نیاز
float calculateMotorRevolutions(float depth, float pitch)
{
  return calculateSpindleRevolutions(depth, pitch)
         * GEAR_RATIO;
}

// تعداد پالس مورد نیاز
long calculatePulses(float depth, float pitch)
{
  float motorRevolutions =
    calculateMotorRevolutions(depth, pitch);

  return (long)(motorRevolutions *
                PULSES_PER_MOTOR_REV);
}

// عمق متناظر با تعداد پالس
float calculateDepthFromPulses(long pulses, float pitch)
{
  if (PULSES_PER_MOTOR_REV <= 0)
    return 0.0;

  float motorRevolutions =
    (float)pulses / PULSES_PER_MOTOR_REV;


  float spindleRevolutions =
    motorRevolutions / GEAR_RATIO;

  return spindleRevolutions * pitch;
}

// =====================================================
// STARTUP ANIMATION
// =====================================================

void startupAnimation()
{
  unsigned long startTime = millis();


  unsigned long lastGearFrame = 0;
  unsigned long lastTextChar = 0;

  byte gearFrame = 0;
  byte textLength = 0;

  const char companyText[] = "KYAN SANAT NOVIN";

  while (millis() - startTime < 5000)
  {
    unsigned long now = millis();

    if (now - lastGearFrame >= 150)
    {
      lastGearFrame = now;

      gearFrame++;

      if (gearFrame >= 4)

        gearFrame = 0;
    }

    if (now - startTime >= 2000)
    {
      if (now - lastTextChar >= 100)
      {
        lastTextChar = now;

        if (textLength < 16)
          textLength++;
      }
    }

    display.clearDisplay();

    display.setTextColor(KS0108_ON);
    display.setTextSize(2);

    display.setCursor(3, 10);
    display.print("K.S.N");


    drawGear(gearFrame);

    if (textLength > 0)
    {
      display.setTextSize(1);

      display.setCursor(15, 48);

      for (byte i = 0; i < textLength; i++)
      {
        display.print(companyText[i]);
      }
    }

    display.display();
  }

  display.clearDisplay();
  display.display();
}

void setDefaultParameters()
{
  parameters.pitch = 1.00;
  parameters.depth = 10.0;

  parameters.speed = 50;

  parameters.torque = 50;
  parameters.torqueDelay = 0.50;

  parameters.forward = 2.0;
  parameters.back = 1.0;


  parameters.chipBreak = 1;

  parameters.tapDirection = 0;
}

// =====================================================
// EEPROM SAVE
// =====================================================

void saveParameters()
{
  EEPROM.put(0, EEPROM_MARK);
  EEPROM.put(4, parameters);

  savedParameters = parameters;
}


// =====================================================
// EEPROM LOAD
// =====================================================

void loadParameters()
{
  int mark;

  EEPROM.get(0, mark);

  if (mark != EEPROM_MARK)
  {
    setDefaultParameters();
    saveParameters();
  }
  else

  {
    EEPROM.get(4, parameters);

    savedParameters = parameters;

    float maxRPM = getMaxSpindleRPM();

    if (parameters.speed > (int)maxRPM)
      parameters.speed = (int)maxRPM;

    if (parameters.depth > MACHINE_MAX_DEPTH)
      parameters.depth = MACHINE_MAX_DEPTH;
  }
}

// =====================================================
// CHANGE PARAMETER
// =====================================================

void changeParameter(int dir)
{
  if (parameterPosition == 0)
  {
    parameters.pitch += 0.01 * dir;


    if (parameters.pitch < 0.01)
      parameters.pitch = 0.01;

    if (parameters.pitch > 20.00)
      parameters.pitch = 20.00;
  }

  else if (parameterPosition == 1)
  {
    parameters.depth += 0.1 * dir;

    if (parameters.depth < 0.1)
      parameters.depth = 0.1;

    if (parameters.depth > MACHINE_MAX_DEPTH)
      parameters.depth = MACHINE_MAX_DEPTH;
  }

  else if (parameterPosition == 2)
  {
    float maxRPM = getMaxSpindleRPM();

    parameters.speed += 10 * dir;

    if (parameters.speed < 10)
      parameters.speed = 10;

    if (parameters.speed > (int)maxRPM)
      parameters.speed = (int)maxRPM;
  }

  else if (parameterPosition == 3)
  {
    parameters.torque += dir;

    if (parameters.torque < 1)
      parameters.torque = 1;

    if (parameters.torque > 100)

      parameters.torque = 100;
  }

  else if (parameterPosition == 4)
  {
    parameters.torqueDelay += 0.01 * dir;

    if (parameters.torqueDelay < 0.01)
      parameters.torqueDelay = 0.01;

    if (parameters.torqueDelay > 10.00)
      parameters.torqueDelay = 10.00;
  }

  else if (parameterPosition == 5)
  {
    parameters.forward += dir;

    if (parameters.forward < 1)
      parameters.forward = 1;

    if (parameters.forward > 20)
      parameters.forward = 20;
  }

  else if (parameterPosition == 6)
  {
    parameters.back += dir;

    if (parameters.back < 1)
      parameters.back = 1;

    if (parameters.back > 20)
      parameters.back = 20;
  }

  else if (parameterPosition == 7)
  {
    parameters.chipBreak = !parameters.chipBreak;
  }

  else if (parameterPosition == 8)
  {
    if (dir > 0)
      parameters.tapDirection = 1;
    else
      parameters.tapDirection = 0;
  }
}

// =====================================================
// EDIT PARAMETER
// =====================================================

void editParameter()
{
  byte key;

  if (digitalRead(KEY_UP) == LOW)
    key = KEY_UP;
  else
    key = KEY_DOWN;

  unsigned long start = millis();
  unsigned long lastChange = 0;

  bool first = true;

  while (digitalRead(key) == LOW)
  {
    unsigned long now = millis();

    unsigned long holdTime = now - start;

    unsigned long interval;

    if (holdTime < 500)
      interval = 300;
    else if (holdTime < 1500)

      interval = 120;
    else
      interval = 50;

    if (first || now - lastChange >= interval)
    {
      lastChange = now;
      first = false;

      if (key == KEY_UP)
        changeParameter(1);
      else
        changeParameter(-1);

      showSettings();
    }
  }
}

// محاسبه فاصله زمانی بین پالس‌ها
unsigned long getPulseIntervalForSpindleRPM(int spindleRPM)
{
  float motorRPM = (float)spindleRPM * GEAR_RATIO;

  if (motorRPM <= 0.0)
    motorRPM = 1.0;

  float pulsesPerSecond =
    (motorRPM * PULSES_PER_MOTOR_REV) / 60.0;

  if (pulsesPerSecond < 1.0)
    pulsesPerSecond = 1.0;


  float halfPeriod =
    1000000.0 / (pulsesPerSecond * 2.0);

  if (halfPeriod < 2.0)
    halfPeriod = 2.0;

  return (unsigned long)halfPeriod;
}

unsigned long getManualPulseInterval()
{
  return getPulseIntervalForSpindleRPM(manualSpeed);
}

unsigned long getAutoPulseInterval()
{
  return getPulseIntervalForSpindleRPM(parameters.speed);
}

// =====================================================
// SERVO START
// =====================================================

void servoStart()
{
  digitalWrite(SERVO_ON_PIN, HIGH);

  servoMoving = true;

  lastPulseTime = micros();

  pulseState = LOW;

  digitalWrite(SERVO_PULSE_PIN, LOW);
}

// =====================================================
// SERVO STOP
// =====================================================

void servoStop()
{
  servoMoving = false;

  pulseState = LOW;

  digitalWrite(SERVO_PULSE_PIN, LOW);

  digitalWrite(SERVO_ON_PIN, LOW);
}


// =====================================================
// MANUAL CW
// =====================================================

void servoCW()
{
  digitalWrite(SERVO_DIR_PIN, LOW);

  if (!servoMoving)
    servoStart();

  unsigned long now = micros();

  if (now - lastPulseTime >= getManualPulseInterval())
  {

    lastPulseTime = now;

    pulseState = !pulseState;

    digitalWrite(SERVO_PULSE_PIN, pulseState);

    if (pulseState == HIGH)
    {
      long maxPulses =
        calculatePulses(
          MACHINE_MAX_DEPTH,
          parameters.pitch
        );

      if (manualPulseCount < maxPulses)
        manualPulseCount++;

      manualDepth =
        calculateDepthFromPulses(
          manualPulseCount,

          parameters.pitch
        );

      if (manualDepth > MACHINE_MAX_DEPTH)
        manualDepth = MACHINE_MAX_DEPTH;

      showManual();
    }
  }
}

// =====================================================
// MANUAL CCW
// =====================================================

void servoCCW()
{
  digitalWrite(SERVO_DIR_PIN, HIGH);

  if (!servoMoving)
    servoStart();

  unsigned long now = micros();

  if (now - lastPulseTime >= getManualPulseInterval())
  {
    lastPulseTime = now;

    pulseState = !pulseState;

    digitalWrite(SERVO_PULSE_PIN, pulseState);

    if (pulseState == HIGH)
    {

      if (manualPulseCount > 0)
        manualPulseCount--;

      manualDepth =
        calculateDepthFromPulses(
          manualPulseCount,
          parameters.pitch
        );

      if (manualDepth < 0.0)
        manualDepth = 0.0;

      showManual();
    }
  }
}

// =====================================================
// MANUAL HANDLE CONTROL

// =====================================================

void manualHandleControl()
{
  bool button6 = (digitalRead(HANDLE_START) == LOW);
  bool button7 = (digitalRead(HANDLE_STOP) == LOW);

  // هر دو همزمان = توقف کامل
  if (button6 && button7)
  {
    servoStop();

    handleWasPressed = true;
    handleReleaseTime = 0;

    return;
  }


  // شستی 6 = افزایش عمق
  if (button6)
  {
    handleWasPressed = true;
    handleReleaseTime = 0;

    servoCW();

    return;
  }

  // شستی 7 = کاهش عمق
  if (button7)
  {
    handleWasPressed = true;
    handleReleaseTime = 0;

    servoCCW();

    return;

  }

  // هیچ شستی فشرده نیست
  servoStop();

  if (handleWasPressed)
  {
    handleWasPressed = false;
    handleReleaseTime = millis();
  }

  // بعد از 3 ثانیه عمق صفر شود
  if (!handleWasPressed &&
      handleReleaseTime != 0 &&
      millis() - handleReleaseTime >= 3000)
  {
    manualDepth = 0.0;
    manualPulseCount = 0;

    handleReleaseTime = 0;

    showManual();
  }
}

// =====================================================
// MANUAL CONTROL
// =====================================================

void manualControl()
{
  if (digitalRead(KEY_MENU) == LOW)
  {
    while (digitalRead(KEY_MENU) == LOW)
      delay(10);

    servoStop();

    manualMode = false;

    manualSpeed = savedParameters.speed;

    pulseState = LOW;

    digitalWrite(SERVO_PULSE_PIN, LOW);

    showMainMenu();

    delay(150);

    return;
  }

  if (digitalRead(KEY_UP) == LOW)
  {
    manualSpeedControl();

    showManual();

    delay(100);

    return;
  }

  if (digitalRead(KEY_DOWN) == LOW)
  {
    manualSpeedControl();

    showManual();

    delay(100);

    return;
  }

  manualHandleControl();
}

// ==================
// AUTO RETURN PULSE GENERATION
// =====================================================

void autoReturnPulse()
{
  // ===================================================
  // CHIP BREAK RETURN
  // ===================================================

  if (parameters.chipBreak &&
      chipBreakReturning)
  {
    digitalWrite(SERVO_DIR_PIN, HIGH);

    if (!servoMoving)
      servoStart();

    unsigned long now = micros();

    if (now - lastPulseTime >= getAutoPulseInterval())
    {
      lastPulseTime = now;

      pulseState = !pulseState;

      digitalWrite(
        SERVO_PULSE_PIN,
        pulseState
      );

      if (pulseState == HIGH)
      {
        long breakPulses =
          calculatePulses(

            parameters.back,
            parameters.pitch
          );

        if (breakPulses < 1)
          breakPulses = 1;

        long returnedPulses =
          chipBreakReturnStartPulse -
          autoPulseCount;

        if (returnedPulses < breakPulses)
        {
          if (autoPulseCount > 0)
            autoPulseCount--;

          autoDepth =
            calculateDepthFromPulses(
              autoPulseCount,
              parameters.pitch
            );


          if (autoDepth < 0.0)
            autoDepth = 0.0;

          showAuto();
        }

        // ---------------------------------------------
        // برگشت Chip Break کامل شد
        // ---------------------------------------------

        returnedPulses =
          chipBreakReturnStartPulse -
          autoPulseCount;

        if (returnedPulses >= breakPulses)
        {
          servoStop();

          chipBreakReturning = false;

          // مرحله Forward بعدی از این نقطه شروع می‌شود
          chipForwardStartPulse =
            autoPulseCount;

          autoState = AUTO_RUNNING;

          showAuto();
        }
      }
    }

    return;
  }

  // ===================================================
  // FINAL RETURN TO ZERO
  // ===================================

// ================

  digitalWrite(SERVO_DIR_PIN, HIGH);

  if (autoPulseCount <= 0)
  {
    autoPulseCount = 0;

    autoDepth = 0.0;

    servoStop();

    autoState = AUTO_FINISHED;

    showAuto();

    return;
  }

  if (!servoMoving)
    servoStart();


  unsigned long now = micros();

  if (now - lastPulseTime >= getAutoPulseInterval())
  {
    lastPulseTime = now;

    pulseState = !pulseState;

    digitalWrite(
      SERVO_PULSE_PIN,
      pulseState
    );

    if (pulseState == HIGH)
    {
      autoPulseCount--;

      if (autoPulseCount < 0)
        autoPulseCount = 0;


      autoDepth =
        calculateDepthFromPulses(
          autoPulseCount,
          parameters.pitch
        );

      if (autoDepth < 0.0)
        autoDepth = 0.0;

      showAuto();

      if (autoPulseCount == 0)
      {
        autoDepth = 0.0;

        servoStop();

        autoState = AUTO_FINISHED;

        showAuto();

      }
    }
  }
}

// =====================================================
// AUTO START
// =====================================================

void startAutoCycle()
{
  if (!startEnabled ||
      autoState != AUTO_READY)
    return;

  startEnabled = false;


  autoState = AUTO_RUNNING;

  autoDepth = 0.0;

  autoPulseCount = 0;

  autoTargetPulses =
    calculatePulses(
      parameters.depth,
      parameters.pitch
    );

  // ==========================================
  // CHIP BREAK INITIALIZATION
  // ==========================================

  chipForwardStartPulse = 0;

  chipBreakReturnStartPulse = 0;

  chipBreakReturning = false;

  autoStartTime = millis();

  servoStop();

  showAuto();
}

// =====================================================
// AUTO STOP
// =====================================================

void stopAutoCycle()
{
  servoStop();

  autoState = AUTO_STOPPED;

  showAuto();
}

// =====================================================
// AUTO RETURN
// =====================================================

void returnAuto()
{
  // اگر قرار است برگشت نهایی انجام شود،
  // Chip Break را لغو می‌کنیم.


  chipBreakReturning = false;

  if (autoPulseCount <= 0)
  {
    autoPulseCount = 0;

    autoDepth = 0.0;

    servoStop();

    autoState = AUTO_FINISHED;

    showAuto();

    return;
  }

  autoState = AUTO_RETURNING;

  servoStop();


  showAuto();
}

// =====================================================
// AUTO FORWARD PULSE GENERATION
// =====================================================

void autoForwardPulse()
{
  // ===================================================
  // حرکت به جلو
  // ===================================================


  digitalWrite(SERVO_DIR_PIN, LOW);

  if (!servoMoving)
    servoStart();

  unsigned long now = micros();

  if (now - lastPulseTime >= getAutoPulseInterval())
  {
    lastPulseTime = now;

    pulseState = !pulseState;

    digitalWrite(
      SERVO_PULSE_PIN,
      pulseState
    );

    if (pulseState == HIGH)

    {
      // افزایش موقعیت فعلی
      if (autoPulseCount < autoTargetPulses)
        autoPulseCount++;

      autoDepth =
        calculateDepthFromPulses(
          autoPulseCount,
          parameters.pitch
        );

      if (autoDepth > parameters.depth)
        autoDepth = parameters.depth;

      showAuto();

      // =================================================
      // رسیدن به عمق نهایی
      // 

// =================================================

      if (autoPulseCount >= autoTargetPulses)
      {
        autoDepth = parameters.depth;

        servoStop();

        // مهم:
        // اگر به عمق نهایی رسیدیم، دیگر Chip Break نداریم
        chipBreakReturning = false;

        returnAuto();

        return;
      }

      // 

// =================================================
      // CHIP BREAK
      // =================================================

      if (parameters.chipBreak)
      {
        long forwardPulses =
          calculatePulses(
            parameters.forward,
            parameters.pitch
          );

        if (forwardPulses < 1)
          forwardPulses = 1;

        // مقدار حرکتی که در مرحله فعلی جلو رفته‌ایم
        long currentForwardPulses =
          autoPulseCount -

          chipForwardStartPulse;

        // ===============================================
        // رسیدن به مقدار FORWARD
        // ===============================================

        if (currentForwardPulses >= forwardPulses)
        {
          chipBreakReturnStartPulse =
            autoPulseCount;

          chipBreakReturning = true;

          servoStop();

          autoState = AUTO_RETURNING;


          showAuto();

          return;
        }
      }
    }
  }
}

// =====================================================
// AUTO CYCLE
// =====================================================

void autoCycle()
{
  if (autoState == AUTO_RUNNING)

  {
    if (torqueLimitReached())
    {
      stopAutoCycle();

      return;
    }

    autoForwardPulse();

    return;
  }

  if (autoState == AUTO_RETURNING)
  {
    autoReturnPulse();

    return;
  }
}

// =====================================================
// AUTO HANDLE CONTROL
// =====================================================

void autoHandleControl()
{
  bool button6 =
    (digitalRead(HANDLE_START) == LOW);

  bool button7 =
    (digitalRead(HANDLE_STOP) == LOW);

  // ===================================================
  // هر دو همزمان = هیچ حرکت
  // 

// ===================================================

  if (button6 && button7)
  {
    servoStop();

    return;
  }

  // ===================================================
  // START
  // ===================================================

  if (button6)
  {
    if (startEnabled &&

        autoState == AUTO_READY)
    {
      startAutoCycle();
    }

    return;
  }

  // ===================================================
  // STOP
  // ===================================================

  if (button7)
  {
    if (autoState == AUTO_RUNNING)
    {
      stopAutoCycle();


      // توقف Chip Break و برگشت کامل
      chipBreakReturning = false;

      returnAuto();

      return;
    }

    else if (autoState == AUTO_STOPPED)
    {
      // برگشت کامل به صفر
      chipBreakReturning = false;

      returnAuto();

      return;
    }

    else if (autoState == AUTO_RETURNING)
    {

      // اگر در حال Chip Break Return هستیم،
      // STOP باید آن را لغو کرده و برگشت کامل انجام دهد

      chipBreakReturning = false;

      returnAuto();

      return;
    }

    return;
  }

  // ===================================================
  // هیچ شستی فشرده نیست
  // ===================================================


  if (stopButtonPressed)
    stopButtonPressed = false;

  // ===================================================
  // آماده شدن برای سیکل بعد
  // ===================================================

  if (autoState == AUTO_FINISHED &&
      !button6 &&
      !button7)
  {
    if (!startEnabled)
    {
      if (millis() - stopButtonTime >= 2000)
      {
        startEnabled = true;


        autoState = AUTO_READY;

        autoDepth = 0.0;

        autoPulseCount = 0;

        chipForwardStartPulse = 0;

        chipBreakReturnStartPulse = 0;

        chipBreakReturning = false;

        showAuto();
      }
    }
    else
    {
      startEnabled = false;

      stopButtonTime = millis();

    }
  }
}

// =====================================================
// AUTO CONTROL
// =====================================================

void autoControl()
{
  if (digitalRead(KEY_MENU) == LOW)
  {
    while (digitalRead(KEY_MENU) == LOW)
      delay(10);

    servoStop();

    autoMode = false;

    autoState = AUTO_READY;

    startEnabled = true;

    autoDepth = 0.0;

    autoPulseCount = 0;

    chipForwardStartPulse = 0;

    chipBreakReturnStartPulse = 0;

    chipBreakReturning = false;

    showMainMenu();

    delay(150);

    return;

  }

  autoHandleControl();

  autoCycle();
}

// =====================================================
// SETTINGS CONTROL
// =====================================================

void settingsControl()
{
  if (editMode)
  {
    if (digitalRead(KEY_UP) == LOW)
    {

      editParameter();

      return;
    }

    if (digitalRead(KEY_DOWN) == LOW)
    {
      editParameter();

      return;
    }

    if (digitalRead(KEY_ENTER) == LOW)
    {
      while (digitalRead(KEY_ENTER) == LOW)
        delay(10);

      editMode = false;

      showSettings();


      delay(150);

      return;
    }

    if (digitalRead(KEY_MENU) == LOW)
    {
      while (digitalRead(KEY_MENU) == LOW)
        delay(10);

      editMode = false;

      parameters = savedParameters;

      showSettings();

      delay(150);

      return;
    }


    return;
  }

  if (digitalRead(KEY_UP) == LOW)
  {
    if (parameterPosition > 0)
      parameterPosition--;

    showSettings();

    delay(150);

    while (digitalRead(KEY_UP) == LOW)
      delay(10);

    return;
  }

  if (digitalRead(KEY_DOWN) == LOW)
  {

    if (parameterPosition < 9)
      parameterPosition++;

    showSettings();

    delay(150);

    while (digitalRead(KEY_DOWN) == LOW)
      delay(10);

    return;
  }

  if (digitalRead(KEY_ENTER) == LOW)
  {
    while (digitalRead(KEY_ENTER) == LOW)
      delay(10);

    if (parameterPosition == 9)
    {
      saveParameters();


      display.clearDisplay();

      display.setTextSize(2);
      display.setTextColor(KS0108_ON);

      display.setCursor(38, 25);
      display.print("SAVED");

      display.display();

      delay(1000);

      showSettings();

      return;
    }

    editMode = true;

    showSettings();


    delay(150);

    return;
  }

  if (digitalRead(KEY_MENU) == LOW)
  {
    while (digitalRead(KEY_MENU) == LOW)
      delay(10);

    settingsMode = false;

    parameters = savedParameters;

    showMainMenu();

    delay(150);

    return;
  }

}

// =====================================================
// MAIN MENU CONTROL
// =====================================================

void mainMenuControl()
{
  if (digitalRead(KEY_UP) == LOW)
  {
    if (menuPosition > 0)
      menuPosition--;

    showMainMenu();

    delay(150);

    while (digitalRead(KEY_UP) == LOW)
      delay(10);

    return;
  }

  if (digitalRead(KEY_DOWN) == LOW)
  {
    if (menuPosition < 2)
      menuPosition++;

    showMainMenu();

    delay(150);

    while (digitalRead(KEY_DOWN) == LOW)
      delay(10);

    return;
  }

  if (digitalRead(KEY_ENTER) == LOW)
  {
    while (digitalRead(KEY_ENTER) == LOW)
      delay(10);

    if (menuPosition == 0)
    {
      autoMode = true;

      autoState = AUTO_READY;

      startEnabled = true;

      autoDepth = 0.0;

      autoPulseCount = 0;

      chipForwardStartPulse = 0;

      chipBreakReturnStartPulse = 0;

      chipBreakReturning = false;

      showAuto();
    }

    else if (menuPosition == 1)
    {
      manualMode = true;

      manualSpeed = savedParameters.speed;

      manualDepth = 0.0;

      manualPulseCount = 0;

      handleWasPressed = false;

      showManual();
    }

    else if (menuPosition == 2)
    {
      settingsMode = true;

      editMode = false;

      parameterPosition = 0;

      parameters = savedParameters;

      showSettings();
    }

    delay(150);

    return;
  }
}

// ===================================

// ==================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(9600);

  // ===================================================
  // کلیدها
  // ===================================================

  pinMode(KEY_UP, INPUT_PULLUP);
  pinMode(KEY_DOWN, INPUT_PULLUP);
  pinMode(KEY_ENTER, INPUT_PULLUP);

  pinMode(KEY_MENU, INPUT_PULLUP);

  pinMode(HANDLE_START, INPUT_PULLUP);
  pinMode(HANDLE_STOP, INPUT_PULLUP);

  // ===================================================
  // سروو
  // ===================================================

  pinMode(SERVO_PULSE_PIN, OUTPUT);
  pinMode(SERVO_DIR_PIN, OUTPUT);

  pinMode(SERVO_ON_PIN, OUTPUT);

  digitalWrite(SERVO_PULSE_PIN, LOW);

  digitalWrite(SERVO_DIR_PIN, LOW);
  digitalWrite(SERVO_ON_PIN, LOW);

  // ===================================================
  // GLCD
  // ===================================================

  if (display.begin(KS0108_CS_ACTIVE_HIGH) == false)
  {
    Serial.println("GLCD initialization failed!");

    while (true)
    {
    }
  }


  display.clearDisplay();
  display.display();

  // ===================================================
  // STARTUP
  // ===================================================

  startupAnimation();

  // ===================================================
  // EEPROM
  // ===================================================


  loadParameters();

  // ===================================================
  // MAIN MENU
  // ===================================================

  showMainMenu();
}

// =====================================================
// LOOP
// =====================================================


void loop()
{
  if (manualMode)
  {
    manualControl();

    return;
  }

  if (autoMode)
  {
    autoControl();

    return;
  }

  if (settingsMode)
  {
    settingsControl();

    return;
  }

  mainMenuControl();
}
